**Goal:** Build a high-converting landing page for Attraction by Video that drives free channel audit bookings from real estate agents, with a secondary direct membership purchase path.

**Architecture:** Single-page Next.js app with component-based sections. Each page section is its own React component. GHL form is embedded via iframe/script. Stripe links are external. ### Typography Rules
- **Headlines:** `font-family: 'Cabinet Grotesk', sans-serif; font-weight: 900; letter-spacing: -0.03em; line-height: 1.05;`
- **Body:** `font-family: 'Satoshi', sans-serif; font-weight: 400; line-height: 1.6; max-width: 65ch;`
- **Labels/Eyebrows:** `font-family: 'Satoshi', sans-serif; font-weight: 600; font-size: 11px; text-transform: uppercase; letter-spacing: 0.15em;`
- **Data/Monospace:** `font-family: 'Geist Mono', monospace;`

### Banned Patterns (DO NOT USE)
- Inter, Roboto, Arial, Helvetica, Open Sans fonts
- Purple/blue AI gradients, neon glows
- Pure `#000000` black anywhere — always use `#1A1A1A` or similar off-black
- Centred hero with text overlaid on dark image
- Emojis — use SVG icons (Phosphor Icons or custom)
- 3-column equal card grids — use asymmetric bento layouts
- `ease-in-out` or `linear` easing — use `var(--ease-out-expo)` or `var(--spring)`
- `h-screen` — use `min-h-[100dvh]`
- Unsplash URLs — use `picsum.photos/seed/{name}/800/600` for placeholders

### Animation Pattern (reuse everywhere)
```css
.fade-up {
  opacity: 0;
  transform: translateY(24px);
  transition: all 0.8s var(--ease-out-expo);
}
.fade-up.visible {
  opacity: 1;
  transform: translateY(0);
}
```

Use Intersection Observer to add `.visible` class when elements enter viewport. Stagger children with `transition-delay: calc(var(--index) * 100ms)`.

### Grain Overlay (apply to body)
```css
body::before {
  content: '';
  position: fixed;
  inset: 0;
  background: url("data:image/svg+xml,..."); /* noise SVG */
  pointer-events: none;
  z-index: 9999;
  opacity: 0.03;
}
```

## Task 2: Reusable Animation Hook & ScrollFadeUp Component

**Files:**
- Create: `app/hooks/useInView.ts`
- Create: `app/components/ScrollFadeUp.tsx`

- [ ] **Step 1: Create useInView hook**

```tsx
'use client'
import { useEffect, useRef, useState } from 'react'

export function useInView(options?: IntersectionObserverInit) {
  const ref = useRef<HTMLDivElement>(null)
  const [isVisible, setIsVisible] = useState(false)

  useEffect(() => {
    const el = ref.current
    if (!el) return

    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        setIsVisible(true)
        observer.unobserve(el)
      }
    }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px', ...options })

    observer.observe(el)
    return () => observer.disconnect()
  }, [])

  return { ref, isVisible }
}
```

- [ ] **Step 2: Create ScrollFadeUp wrapper component**

```tsx
'use client'
import { useInView } from '../hooks/useInView'

interface ScrollFadeUpProps {
  children: React.ReactNode
  delay?: number
  className?: string
}

export function ScrollFadeUp({ children, delay = 0, className = '' }: ScrollFadeUpProps) {
  const { ref, isVisible } = useInView()

  return (
    <div
      ref={ref}
      className={`fade-up ${isVisible ? 'visible' : ''} ${className}`}
      style={{ transitionDelay: `${delay}ms` }}
    >
      {children}
    </div>
  )
}
```

// Count from 0 to target over ~2 seconds
const [count, setCount] = useState(0)
const { ref, isVisible } = useInView()

useEffect(() => {
  if (!isVisible) return
  const target = 287400
  const duration = 2000
  const steps = 60
  const increment = target / steps
  let current = 0
  const timer = setInterval(() => {
    current += increment
    if (current >= target) {
      setCount(target)
      clearInterval(timer)
    } else {
      setCount(Math.floor(current))
    }
  }, duration / steps)
  return () => clearInterval(timer)
}, [isVisible])
